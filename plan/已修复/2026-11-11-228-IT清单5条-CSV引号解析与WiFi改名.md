```javascript
// N217 CSV 引号解析
function splitCsvRow(line) {
  const s = String(line == null ? '' : line);
  const out = []; let cell = '', inQuote = false;
  for (let i = 0; i < s.length; i += 1) {
    const ch = s.charAt(i);
    if (inQuote) {
      if (ch === '"') { if (s.charAt(i+1) === '"') { cell += '"'; i += 1; } else inQuote = false; }
      else cell += ch;
    } else if (ch === '"' && cell === '') inQuote = true;
    else if (ch === ',') { out.push(cell); cell = ''; }
    else cell += ch;
  }
  out.push(cell);
  return out;
}
```
