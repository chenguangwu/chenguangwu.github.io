function __XResText(){
    var r=RESEXPR;
    if(!r) return '';
    var t=(r.textContent||'').replace(/\s+/g,' ').trim();
    if(!t||t==='--'||t==='—'||t==='-') return '';
    if(/请输入|无效|不正确|错误|Error|NaN/.test(t)) return '';
    return t.slice(0,80);
}
function __XTime(t){
    var d=new Date(t);
    function q(x){return (x<10?'0':'')+x;}
    return q(d.getMonth()+1)+'-'+q(d.getDate())+' '+q(d.getHours())+':'+q(d.getMinutes());
}
function __XRender(keys,rk,en){
    try{
        var arr=TBKit.history.list().slice(0,10);
        var head=[en?'Time':'时间'].concat(keys).concat([rk]);
        var rows=arr.map(function(it){
            var row=[__XTime(it.t)];
            for(var i=0;i<keys.length;i++){var v=it.d[keys[i]];row.push(v==null?'':v);}
            var r=it.d[rk];row.push(r==null?'':r);
            return row;
        });
        TBKit.matrix.table('#XTable',{head:head,rows:rows});
        TBKit.util.diffBox('#XDiff');
        TBKit.history.mount('#XHist',function(rec){
            __X_skip=true;
            try{
                for(var k in rec){
                    if(k===rk) continue;
                    var e=document.getElementById(k);
                    if(!e) continue;
                    var t=(e.type||'').toLowerCase();
                    if(t==='checkbox'||t==='radio'){e.checked=(String(e.value)===String(rec[k])||rec[k]==='1'||rec[k]===true);}
                    else{e.value=rec[k];}
                }
                CALC();
            }catch(e2){}
            __X_skip=false;
        });
        var nums=[],labels=[];
        for(var j=arr.length-1;j>=0;j--){
            var v=parseFloat(arr[j].d[rk]);
            if(isFinite(v)){nums.push(v);labels.push(__XTime(arr[j].t));}
        }
        if(nums.length>=2){
            TBKit.chart.line('#XChart',{labels:labels,series:[{name:en?'Result trend':'结果趋势',data:nums}]});
        }
    }catch(e3){}
}
function __XExtras(){
    if (typeof TBKit === 'undefined') return;
    if (__X_skip) return;
    var snap=__XSnap();
    var keys=Object.keys(snap);
    if(!keys.length) return;
    if(keys.length>12) keys=keys.slice(0,12);
    var res=__XResText();
    if(!res) return;
    var en=TBKit.util.isEn();
    var rk=en?'Result':'结果';
    var now=Date.now();
    if(now-__X_last<1000) return;
    var last=null;
    try{ last=TBKit.history.last(); }catch(e){}
    if(last){
        var same=true;
        for(var i=0;i<keys.length;i++){
            if(String(last[keys[i]])!==String(snap[keys[i]])){same=false;break;}
        }
        if(same&&String(last[rk])===String(res)) return;
    }
    __X_last=now;
    var rec={};
    for(var i=0;i<keys.length;i++) rec[keys[i]]=snap[keys[i]];
    rec[rk]=res;
    TBKit.history.record(rec);
    __XRender(keys,rk,en);
}
function __XExport(){
    if (typeof TBKit === 'undefined') return;
    try { TBKit.util.exportTable('#XTable','X.csv'); } catch(e){}
}
